  carMarker.visible=state.cameraMode!==0;carMarker.position.copy(car.pos);carMarker.position.y=.42;carMarker.rotation.y=state.heading;
  $('speed').textContent=Math.round(Math.abs(state.speed)*3.6);
}

function animate(){
  requestAnimationFrame(animate);if(!state.ready)return;
  const dt=Math.min(.05,clock.getDelta());updateCar(dt);updateCamera(dt);renderer.render(scene,camera);
}

function bindHold(id,key){
  const el=$(id),down=e=>{e.preventDefault();state[key]=true;el.classList.add('down');},up=e=>{if(e)e.preventDefault();state[key]=false;el.classList.remove('down');};
  el.addEventListener('pointerdown',down);el.addEventListener('pointerup',up);el.addEventListener('pointercancel',up);el.addEventListener('pointerleave',e=>{if(e.buttons===0)up(e)});
}
function addEventListeners(){
  bindHold('gas','throttle');bindHold('brake','brake');bindHold('left','steerL');bindHold('right','steerR');
  addEventListener('keydown',e=>{
    if(['ArrowUp','w','W'].includes(e.key))state.throttle=true;if(['ArrowDown','s','S'].includes(e.key))state.brake=true;if(['ArrowLeft','a','A'].includes(e.key))state.steerL=true;if(['ArrowRight','d','D'].includes(e.key))state.steerR=true;
    if(e.key==='c'||e.key==='C')cycleCamera();if(e.key==='r'||e.key==='R')resetToRoad();
  });
  addEventListener('keyup',e=>{if(['ArrowUp','w','W'].includes(e.key))state.throttle=false;if(['ArrowDown','s','S'].includes(e.key))state.brake=false;if(['ArrowLeft','a','A'].includes(e.key))state.steerL=false;if(['ArrowRight','d','D'].includes(e.key))state.steerR=false;});
  let drag=false,lastX=0,lastY=0;
  renderer.domElement.addEventListener('pointerdown',e=>{if(e.target!==renderer.domElement)return;drag=true;lastX=e.clientX;lastY=e.clientY;renderer.domElement.setPointerCapture?.(e.pointerId);});
  renderer.domElement.addEventListener('pointermove',e=>{if(!drag||state.cameraMode!==0)return;const dx=e.clientX-lastX,dy=e.clientY-lastY;lastX=e.clientX;lastY=e.clientY;state.lookYaw-=dx*.0045;state.lookPitch=THREE.MathUtils.clamp(state.lookPitch+dy*.0035,-.5,.42);});
  renderer.domElement.addEventListener('pointerup',()=>drag=false);renderer.domElement.addEventListener('pointercancel',()=>drag=false);
  $('autoBtn').onclick=()=>{state.auto=!state.auto;if(state.auto){const n=nearestOnRoute(car.pos);state.autoDist=n.along;state.speed=Math.max(2,state.speed);$('autoBtn').classList.add('active');toast('Automatische Fahrt aktiviert');}else{$('autoBtn').classList.remove('active');toast('Manuelle Fahrt aktiviert');}};
  $('cameraBtn').onclick=cycleCamera;$('resetBtn').onclick=resetToRoad;
  $('settingsBtn').onclick=()=>$('drawer').classList.toggle('open');
  $('autoSpeed').oninput=e=>{state.autoSpeedKmh=Number(e.target.value);$('autoSpeedLabel').textContent=state.autoSpeedKmh+' km/h';};
  $('satBtn').onclick=()=>{state.satellite=!state.satellite;tileGroup.visible=state.satellite;$('satBtn').textContent='Satellitenboden: '+(state.satellite?'an':'aus');};
  $('shadowsBtn').onclick=()=>{state.shadows=!state.shadows;renderer.shadowMap.enabled=state.shadows;sun.castShadow=state.shadows;$('shadowsBtn').textContent='Schatten: '+(state.shadows?'an':'aus');};
  $('realityBtn').onclick=()=>{const url=`https://www.google.com/maps/@?api=1&map_action=pano&viewpoint=${state.carLat.toFixed(6)},${state.carLon.toFixed(6)}`;window.open(url,'_blank','noopener');};
  addEventListener('resize',()=>{camera.aspect=innerWidth/innerHeight;camera.updateProjectionMatrix();renderer.setSize(innerWidth,innerHeight);});
}
function cycleCamera(){state.cameraMode=(state.cameraMode+1)%3;state.lookYaw=state.lookPitch=0;toast(['Fahrerperspektive','Verfolgerkamera','Seitenkamera'][state.cameraMode]);}

async function boot(){
  try{
    initThree();animate();
    await loadRoute();buildRoad();
    await Promise.all([loadBuildingsAndTrees(),buildSatellite()]);
    setLoad('Startposition wird gesetzt …',91);placeCarAtStart();
    $('dataInfo').textContent=`Straße: ${state.routeSource} · Gebäude: ${state.buildingSource} · Strecke: ${state.routeKm.toFixed(1)} km`;
    if(state.routeSource==='OpenStreetMap'&&state.buildingSource==='OpenStreetMap')setStatus('ok','Reale Geometrie geladen');else setStatus('bad','Teilweise Näherungsdaten');
    setLoad('Fahrbereit',100);state.ready=true;clock.start();setTimeout(()=>$('loading').style.display='none',350);setTimeout(()=>$('help').style.opacity='0',9000);
  }catch(e){
    console.error(e);setStatus('bad','Fehler beim Aufbau');loadingText.textContent='Die Szene konnte nicht vollständig aufgebaut werden: '+(e?.message||e)+'. Bitte die Seite mit Internetverbindung neu laden.';progress.style.width='100%';progress.style.background='#b64e4e';
  }
}
boot();
