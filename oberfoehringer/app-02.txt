  const pos=[],uv=[],idx=[];
  let dist=0;
  for(let i=0;i<points.length;i++){
    const p=points[i],prev=points[Math.max(0,i-1)],next=points[Math.min(points.length-1,i+1)];
    const dir=new THREE.Vector3().subVectors(next,prev).setY(0).normalize();
    const n=new THREE.Vector3(-dir.z,0,dir.x);
    const l=p.clone().addScaledVector(n,width/2),r=p.clone().addScaledVector(n,-width/2);l.y=r.y=y;
    if(i>0)dist+=p.distanceTo(points[i-1]);
    pos.push(l.x,l.y,l.z,r.x,r.y,r.z);uv.push(0,dist/8,1,dist/8);
    if(i<points.length-1){const a=i*2;idx.push(a,a+2,a+1,a+1,a+2,a+3);}
  }
  const g=new THREE.BufferGeometry();g.setAttribute('position',new THREE.Float32BufferAttribute(pos,3));g.setAttribute('uv',new THREE.Float32BufferAttribute(uv,2));g.setIndex(idx);g.computeVertexNormals();
  const m=new THREE.Mesh(g,material);m.receiveShadow=true;return m;
}

function buildRoad(){
  setLoad('Fahrbahn, Gehwege und Markierungen werden aufgebaut …',28);
  const sidewalkMat=new THREE.MeshStandardMaterial({color:0xb5b3aa,roughness:.95});
  const asphalt=new THREE.MeshStandardMaterial({color:0x2e3031,roughness:.94});
  roadGroup.add(makeRoadStrip(routeWorld,12,.01,sidewalkMat));
  roadGroup.add(makeRoadStrip(routeWorld,7.4,.035,asphalt));
  const dashMat=new THREE.MeshStandardMaterial({color:0xf3f1e5,roughness:.8});
  for(let d=8;d<routeTotal-8;d+=18){
    const a=pointAtDistance(d),b=pointAtDistance(Math.min(d+4,routeTotal));
    const mid=a.clone().lerp(b,.5),len=a.distanceTo(b),dir=new THREE.Vector3().subVectors(b,a);const yaw=Math.atan2(dir.x,dir.z);
    const m=new THREE.Mesh(new THREE.BoxGeometry(.12,.018,len),dashMat);m.position.copy(mid);m.position.y=.065;m.rotation.y=yaw;roadGroup.add(m);
  }
  // Roadside grass strips subtly soften satellite/ground seams.
  const vergeMat=new THREE.MeshStandardMaterial({color:0x65785d,roughness:1,transparent:true,opacity:.72});
  const outer=makeRoadStrip(routeWorld,18,-.01,vergeMat);outer.renderOrder=-1;roadGroup.add(outer);
  roadGroup.children[0].renderOrder=0;roadGroup.children[1].renderOrder=2;
}

function pointAtDistance(d){
  d=Math.max(0,Math.min(routeTotal,d));
  let lo=0,hi=routeCum.length-1;
  while(lo<hi){const m=(lo+hi)>>1;if(routeCum[m]<d)lo=m+1;else hi=m;}
  const i=Math.max(1,lo),d0=routeCum[i-1],d1=routeCum[i],t=d1===d0?0:(d-d0)/(d1-d0);
  return routeWorld[i-1].clone().lerp(routeWorld[i],t);
}
function headingAtDistance(d){
  const a=pointAtDistance(Math.max(0,d-2)),b=pointAtDistance(Math.min(routeTotal,d+2));
  const v=b.sub(a);return Math.atan2(v.x,v.z);
}

function nearestOnRoute(p){
  let best={dist:Infinity,index:0,t:0,point:routeWorld[0],along:0};
  const start=Math.max(0,car.lastRoadIndex-20),end=Math.min(routeWorld.length-2,car.lastRoadIndex+40);
  const ranges = (end-start<5)?[[0,routeWorld.length-2]]:[[start,end]];
  for(const [s,e] of ranges){
    for(let i=s;i<=e;i++){
      const a=routeWorld[i],b=routeWorld[i+1],ab=b.clone().sub(a),den=ab.lengthSq();
      const t=den?THREE.MathUtils.clamp(p.clone().sub(a).dot(ab)/den,0,1):0;
      const q=a.clone().addScaledVector(ab,t),di=q.distanceTo(p);
      if(di<best.dist) best={dist:di,index:i,t,point:q,along:routeCum[i]+a.distanceTo(q)};
    }
  }
  car.lastRoadIndex=best.index;return best;
}
